 21def compute_alignments(log: pd.DataFrame, model: Tuple[PetriNet, Marking, Marking]) -> List[Dict[str, Any]]:
 22    """Computes optimal alignments between the log and the model (Petri Net).
 23
 24    Ordering matters: pm4py does NOT guarantee that the returned alignment list follows
 25    the case order of ``log[...].unique()`` (passing a DataFrame sorts cases by id
 26    internally). Every consumer here pairs alignments to cases positionally, so we convert
 27    to an EventLog once, key each alignment by its case id, and re-emit in appearance
 28    order — making the positional pairing correct regardless of pm4py's internal ordering.
 29
 30    Args:
 31        log (pd.DataFrame): Event log.
 32        model (Tuple[PetriNet, Marking, Marking]): Petri net model (net, im, fm).
 33
 34    Returns:
 35        List[Dict[str, Any]]: The standard PM4Py alignment output (list of dictionaries),
 36        ordered to match ``log[case_id].unique()`` (first-appearance order).
 37    """
 38    case_id_key = "case:concept:name"
 39    (net, im, fm) = model
 40    event_log = pm4py.convert_to_event_log(log)
 41    aligns = pm4py.conformance_diagnostics_alignments(event_log, net, im, fm)
 42    by_case = {trace.attributes["concept:name"]: a for trace, a in zip(event_log, aligns)}
 43    order = list(dict.fromkeys(log[case_id_key]))  # first-appearance order == unique()
 44    return [by_case[cid] for cid in order]

 47def generate_prefixes(
 48    log: pd.DataFrame,
 49    min_length: int = 1,
 50    max_length: Optional[int] = None,
 51    step: int = 1,
 52    gate_activities: Optional[List[str]] = None,
 53    case_id_key: str = "case:concept:name",
 54    timestamp_key: str = "time:timestamp",
 55    activity_key: str = "concept:name",
 56) -> Tuple[pd.DataFrame, pd.DataFrame]:
 57    """Expand each case into varying-length prefix pseudo-cases (LL-07.1).
 58
 59    For each case and each k in range(min_length, max_length+1, step) with k <= case length,
 60    the first k events (time-ordered) become a pseudo-case whose id is f"{orig}@@k{k}".
 61    This lets all existing feature/alignment/label code (which keys off case_id_key) run
 62    unchanged — each prefix simply looks like another case.
 63
 64    Leakage gating: if gate_activities is given (PROCESS_OUTCOME), only prefixes that end
 65    strictly before the first occurrence of any gate activity are emitted, so the outcome is
 66    never already inside the prefix.
 67
 68    Args:
 69        log (pd.DataFrame): The original event log.
 70        min_length (int, optional): Minimum prefix length. Defaults to 1.
 71        max_length (Optional[int], optional): Maximum prefix length. Defaults to None.
 72        step (int, optional): Step size between prefix lengths. Defaults to 1.
 73        gate_activities (Optional[List[str]], optional): Activities that gate prefix generation. Defaults to None.
 74        case_id_key (str, optional): Key for case ID. Defaults to "case:concept:name".
 75        timestamp_key (str, optional): Key for timestamp. Defaults to "time:timestamp".
 76        activity_key (str, optional): Key for activity name. Defaults to "concept:name".
 77
 78    Returns:
 79        Tuple[pd.DataFrame, pd.DataFrame]: A tuple containing:
 80            prefix_log: event-level DataFrame, case_id_key replaced by the unique prefix id,
 81                every other column (events, case attributes, timestamp) preserved.
 82            prefix_index: DataFrame [prefix_id, orig_case_id, prefix_length].
 83    """
 84    # Order events within each case; mergesort keeps original order for equal timestamps.
 85    g = log.sort_values([case_id_key, timestamp_key], kind="mergesort").copy()
 86    g["_ev"] = g.groupby(case_id_key).cumcount()          # 0-based event position
 87    case_len = g.groupby(case_id_key)["_ev"].max() + 1     # Series indexed by case id
 88
 89    # Per-case cap on prefix length. With gating, cap at the position of the first outcome
 90    # event: a length-k prefix covers events 0..k-1, so k <= first_pos excludes the outcome.
 91    if gate_activities:
 92        is_gate = g[activity_key].isin(set(gate_activities))
 93        gate_pos = g.loc[is_gate].groupby(case_id_key)["_ev"].min()
 94        cap = case_len.copy()
 95        cap.loc[gate_pos.index] = gate_pos
 96    else:
 97        cap = case_len
 98
 99    # Build the set of valid (case, k) pairs.
100    specs = []
101    for case_id, c_cap in cap.items():
102        hi = int(c_cap) if max_length is None else min(int(max_length), int(c_cap))
103        for k in range(min_length, hi + 1, step):
104            specs.append((case_id, k))
105
106    if not specs:
107        return (
108            pd.DataFrame(columns=list(log.columns)),
109            pd.DataFrame(columns=["prefix_id", "orig_case_id", "prefix_length"]),
110        )
111
112    spec = pd.DataFrame(specs, columns=[case_id_key, "_k"])
113
114    # Expand: each event joins every valid k of its case, keep events strictly inside the prefix.
115    merged = g.merge(spec, on=case_id_key)
116    merged = merged[merged["_ev"] < merged["_k"]].copy()
117
118    merged["orig_case_id"] = merged[case_id_key]
119    merged[case_id_key] = merged[case_id_key].astype(str) + "@@k" + merged["_k"].astype(str)
120
121    prefix_index = (
122        merged[[case_id_key, "orig_case_id", "_k"]]
123        .drop_duplicates()
124        .rename(columns={case_id_key: "prefix_id", "_k": "prefix_length"})
125        .reset_index(drop=True)
126    )
127    # Keep prefix_length out of the event log so it can never leak into the feature matrix.
128    prefix_log = merged.drop(columns=["_ev", "_k", "orig_case_id"]).reset_index(drop=True)
129    return prefix_log, prefix_index

132def extract_remaining_time(
133    full_log: pd.DataFrame,
134    prefix_log: pd.DataFrame,
135    prefix_index: pd.DataFrame,
136    case_id_key: str = "case:concept:name",
137    timestamp_key: str = "time:timestamp",
138) -> pd.Series:
139    """Per-prefix remaining time in seconds — the regression label for REMAINING_TIME.
140
141    Unlike the classification targets (one label per case, broadcast to all prefixes),
142    remaining time differs for every prefix: it is the time from the END of the observed
143    prefix to the completion of the original case.
144
145        remaining = (last timestamp of the FULL original case)
146                  - (last timestamp inside the prefix)
147
148    The case-end is taken from full_log so the future the model must predict is never
149    inside the features; "now" is the prefix's last event.
150
151    Args:
152        full_log (pd.DataFrame): Complete event log (original case ids).
153        prefix_log (pd.DataFrame): Prefix event log from generate_prefixes; its case_id_key column
154            already holds the unique prefix id.
155        prefix_index (pd.DataFrame): [prefix_id, orig_case_id, prefix_length] from generate_prefixes.
156        case_id_key (str, optional): Key for case ID. Defaults to "case:concept:name".
157        timestamp_key (str, optional): Key for timestamp. Defaults to "time:timestamp".
158
159    Returns:
160        pd.Series: Series indexed by prefix_id, name "remaining_time_seconds" (float).
161    """
162    # Completion time of each ORIGINAL case.
163    case_end = full_log.groupby(case_id_key)[timestamp_key].max()
164    # "Now": last event timestamp inside each prefix (case_id_key == prefix id here).
165    prefix_now = prefix_log.groupby(case_id_key)[timestamp_key].max()
166
167    idx = prefix_index.set_index("prefix_id")
168    end_for_prefix = idx["orig_case_id"].map(case_end)        # indexed by prefix_id
169    remaining = (end_for_prefix - prefix_now).dt.total_seconds()
170    return remaining.rename("remaining_time_seconds")

173def extract_temporal_features(
174    prefix_log: pd.DataFrame,
175    case_id_key: str = "case:concept:name",
176    timestamp_key: str = "time:timestamp",
177) -> pd.DataFrame:
178    """Per-prefix temporal predictors for remaining-time prediction.
179
180    These are the standard, high-impact time signals for remaining-time models.
181
182    Args:
183        prefix_log (pd.DataFrame): Prefix event log.
184        case_id_key (str, optional): Key for case ID. Defaults to "case:concept:name".
185        timestamp_key (str, optional): Key for timestamp. Defaults to "time:timestamp".
186
187    Returns:
188        pd.DataFrame: DataFrame indexed by prefix id with columns:
189            tf_elapsed_s: time from the prefix's first to last event (how long it has run)
190            tf_since_prev_s: gap between the last two events of the prefix (0 if one event)
191            tf_n_events: number of events observed so far
192    """
193    g = prefix_log.sort_values([case_id_key, timestamp_key], kind="mergesort").copy()
194    grp_ts = g.groupby(case_id_key)[timestamp_key]
195
196    elapsed = (grp_ts.max() - grp_ts.min()).dt.total_seconds()
197    n_events = grp_ts.size().astype(float)
198    # diff() yields NaT for the first event of each prefix; the LAST gap is the most
199    # recent inter-event time. fillna(0) covers single-event prefixes.
200    g["_gap_s"] = grp_ts.diff().dt.total_seconds()
201    since_prev = g.groupby(case_id_key)["_gap_s"].last().fillna(0.0)
202
203    return pd.DataFrame({
204        "tf_elapsed_s": elapsed,
205        "tf_since_prev_s": since_prev,
206        "tf_n_events": n_events,
207    })

210def real_deviation_cost(alignment_moves: List[Tuple[Any, Any]]) -> int:
211    """Count the REAL conformance deviations in one alignment.
212
213    A pm4py alignment move is a (log_move, model_move) tuple:
214        (a, a)        synchronous move                                         -> 0 (conforming)
215        (a, '>>')     log move: event a not allowed by the model here         -> +1
216        ('>>', a)     model move on a VISIBLE transition a (expected, missing) -> +1
217        ('>>', None)  model move on an invisible/silent (tau) transition       -> 0 (routing)
218
219    Silent/tau moves are normal routing in (e.g.) Inductive-Miner models and carry zero
220    cost; counting them as deviations is exactly the bug this helper avoids. Mirrors the
221    move classification used for DIAGNOSTICS features in extract_features.
222
223    Args:
224        alignment_moves (List[Tuple[Any, Any]]): List of alignment moves.
225
226    Returns:
227        int: Total cost of real deviations.
228    """
229    cost = 0
230    for log_move, model_move in alignment_moves:
231        if log_move == model_move and log_move != '>>':
232            continue                                   # synchronous move
233        elif model_move == '>>' and log_move is not None:
234            cost += 1                                  # real log move (a, '>>')
235        elif log_move == '>>' and model_move is not None:
236            cost += 1                                  # real visible model move ('>>', a)
237        # else: invisible/tau model move ('>>', None) -> not a deviation
238    return cost

241def extract_label(
242    log: pd.DataFrame,
243    target_type: PredictionTarget,
244    outcome_activities: Optional[List[str]] = None,
245    alignments: Optional[List[Dict[str, Any]]] = None,
246    deviation_cost_threshold: int = 0,
247) -> pd.Series:
248    """Derives labels for each case.
249
250    - PROCESS_OUTCOME: Checks if any activity in 'outcome_activities' exists in the trace.
251    - FUTURE_DEVIATIONS: label = 1 if the case's real deviation cost exceeds
252      deviation_cost_threshold (HL-08). Real cost ignores silent/tau routing moves.
253    - REMAINING_TIME: per-prefix regression label (see extract_remaining_time).
254
255    Args:
256        log (pd.DataFrame): Event log.
257        target_type (PredictionTarget): Type of prediction target.
258        outcome_activities (Optional[List[str]], optional): Activities defining outcomes. Defaults to None.
259        alignments (Optional[List[Dict[str, Any]]], optional): Alignment results. Defaults to None.
260        deviation_cost_threshold (int, optional): Threshold for future deviations. Defaults to 0.
261
262    Raises:
263        ValueError: If required arguments for target types are missing.
264        NotImplementedError: If target type is not supported.
265
266    Returns:
267        pd.Series: Series of labels indexed by case ID.
268    """
269    case_id_key = 'case:concept:name'
270    activity_key = 'concept:name'
271    timestamp_key = 'time:timestamp'
272
273    if target_type == config.PredictionTarget.PROCESS_OUTCOME:
274        if not outcome_activities:
275            raise ValueError("Process Outcome target requires a list of target activities.")
276        
277        outcome_set = set(outcome_activities)
278        
279        def determine_outcome(activities: pd.Series) -> str:
280            # Find intersection between trace activities and target outcomes
281            present_outcomes = set(activities).intersection(outcome_set)
282            
283            if len(present_outcomes) == 1:
284                return present_outcomes.pop() # Exactly one outcome found
285            else:
286                return "uncertain_outcome" # 0 or >1 outcomes found
287                
288        # Group by case and apply the outcome logic
289        return log.groupby(case_id_key)[activity_key].apply(determine_outcome)
290
291    elif target_type == config.PredictionTarget.FUTURE_DEVIATIONS:
292        if alignments is None:
293            raise ValueError("Future Deviations target requires alignment results.")
294        case_ids = log[case_id_key].unique()
295        
296        if len(case_ids) != len(alignments):
297            raise ValueError("Number of alignments does not match number of unique cases in the log.")
298            
299        # HL-08: label = 1 if the case's REAL deviation cost exceeds the threshold.
300        # real_deviation_cost ignores silent/tau routing moves, so a conforming case that
301        # merely skips an optional activity is correctly labeled 0. T defaults to 0
302        # ("any real deviation"); a higher T tolerates minor deviations.
303        deviations = [
304            int(real_deviation_cost(a.get('alignment', [])) > deviation_cost_threshold)
305            for a in alignments
306        ]
307        return pd.Series(deviations, index=case_ids, name="future_deviations")
308
309    elif target_type == config.PredictionTarget.REMAINING_TIME:
310        # Remaining time is a PER-PREFIX regression label (each prefix has a different
311        # remaining time), not a per-case label, so it is produced by
312        # extract_remaining_time() on the prefix path rather than here.
313        raise ValueError(
314            "REMAINING_TIME labels are computed per prefix via extract_remaining_time(); "
315            "route this target through extract_target_and_feature_over_prefixes."
316        )
317
318    else:
319        raise NotImplementedError(f"Target type {target_type} is not supported.")
320        
321# Standard XES event-level / system columns that are never case attributes,
322# even when they happen to be constant within a case (e.g. an all-"complete"
323# lifecycle:transition). Used to filter the constant-per-case detection below.
324_EVENT_LEVEL_COLS = {
325    "concept:name", "time:timestamp", "start_timestamp",
326    "lifecycle:transition", "lifecycle:state",
327    "org:resource", "org:group", "org:role",
328    "@@index", "@@case_index",
329}

332def list_case_attributes(log: pd.DataFrame) -> List[str]:
333    """Return the case-level (trace) attribute column names present in the log.
334
335    A column is treated as case-level if it is either explicitly prefixed
336    'case:' (as PM4Py does for CSV logs) or — for XES logs, where PM4Py
337    flattens trace attributes into unprefixed columns mixed with event
338    attributes — its value is constant within every case. Standard event/
339    system columns (concept:name, time:timestamp, lifecycle:transition, ...)
340    are always excluded.
341
342    Args:
343        log (pd.DataFrame): Event log.
344
345    Returns:
346        List[str]: List of case-level attribute column names.
347    """
348    case_id_key = 'case:concept:name'
349    cols = [c for c in log.columns if c != case_id_key]
350
351    # nunique per case, computed in a single groupby pass.
352    per_case_nunique = log.groupby(case_id_key).nunique(dropna=True)
353
354    keep = set()
355    for c in cols:
356        if c.startswith('case:'):
357            keep.add(c)                       # explicitly a case attribute
358        elif c in _EVENT_LEVEL_COLS:
359            continue                          # known event/system column
360        elif c in per_case_nunique.columns and per_case_nunique[c].max() <= 1:
361            keep.add(c)                       # constant within every case -> trace attribute
362
363    return [c for c in cols if c in keep]     # preserve original column order

366def summarize_case_attributes(log: pd.DataFrame) -> pd.DataFrame:
367    """Small summary table for CLI display of case-level attributes.
368
369    Helps the analyst choose meaningful attributes (HL-03/HL-04).
370
371    Args:
372        log (pd.DataFrame): Event log.
373
374    Returns:
375        pd.DataFrame: Summary table with columns [attribute, dtype, n_unique, example].
376    """
377    case_id_key = 'case:concept:name'
378    rows = []
379    for col in list_case_attributes(log):
380        per_case = log.groupby(case_id_key)[col].first()
381        example = per_case.dropna().iloc[0] if per_case.notna().any() else ""
382        rows.append({
383            "attribute": col,
384            "dtype": str(per_case.dtype),
385            "n_unique": int(per_case.nunique(dropna=True)),
386            "example": example,
387        })
388    return pd.DataFrame(rows, columns=["attribute", "dtype", "n_unique", "example"])

391def extract_features(
392    log: pd.DataFrame,
393    feature_type: PredictionFeature,
394    model: Optional[Tuple] = None,
395    alignments: Optional[List[Dict[str, Any]]] = None,
396    selected_case_attributes: Optional[List[str]] = None,
397) -> pd.DataFrame:
398    """Extracts features based on the specified feature type.
399
400    - CONTROL_FLOW: Extracts the trace prefix (Bag-of-activities).
401    - DIAGNOSTICS: Extracts the alignment-based trace prefix diagnostics (Bag-of-move-types).
402    - CASE_ATTRIBUTES: Extracts static attributes about the case.
403    - HYBRID: Combines the active feature sets.
404
405    Args:
406        log (pd.DataFrame): Event log.
407        feature_type (PredictionFeature): Type of features to extract.
408        model (Optional[Tuple], optional): Process model. Defaults to None.
409        alignments (Optional[List[Dict[str, Any]]], optional): Alignment results. Defaults to None.
410        selected_case_attributes (Optional[List[str]], optional): Subset of case attributes. Defaults to None.
411
412    Raises:
413        ValueError: If required data for feature type is missing or if feature type is unsupported.
414
415    Returns:
416        pd.DataFrame: Feature matrix indexed by case ID.
417    """
418    case_id_key = 'case:concept:name'
419    activity_key = 'concept:name'
420    
421    feature_matrices = []
422    
423    # CONTROL_FLOW: Bag-of-Activities
424    if feature_type in [config.PredictionFeature.CONTROL_FLOW, config.PredictionFeature.HYBRID]:
425        # pd.crosstab -> frequency matrix (rows=cases, cols=activities)
426        cf_df = pd.crosstab(log[case_id_key], log[activity_key])
427        
428        # Prefix columns to avoid naming collisions with case attributes later
429        cf_df.columns = [f"cf_act_{col}" for col in cf_df.columns]
430        feature_matrices.append(cf_df)
431
432    # DIAGNOSTICS: Bag-of-[activity x move_type]
433    if feature_type in [config.PredictionFeature.DIAGNOSTICS, config.PredictionFeature.HYBRID]:
434        if alignments is None:
435            raise ValueError(f"{feature_type.name} requires alignment results.")
436            
437        case_ids = log[case_id_key].unique()
438        if len(case_ids) != len(alignments):
439            raise ValueError("Number of alignments does not match unique cases in log.")
440            
441        diag_records = []
442        for case_id, align_dict in zip(case_ids, alignments):
443            record = {'case_id': case_id}
444            alignment_moves = align_dict.get('alignment', [])
445            
446            for log_move, model_move in alignment_moves:
447                # Determine the specific move type and activity
448                if log_move == model_move and log_move != '>>':
449                    feat_name = f"sync_move_{log_move}"
450                elif model_move == '>>' and log_move != None:
451                    feat_name = f"log_move_{log_move}"
452                elif log_move == '>>' and model_move != None:
453                    feat_name = f"model_move_{model_move}"
454                else:
455                    # Invisible transitions
456                    continue 
457                
458                record[feat_name] = record.get(feat_name, 0) + 1
459                
460            diag_records.append(record)
461
462        diag_df = pd.DataFrame(diag_records).set_index('case_id').fillna(0).astype(int)
463        feature_matrices.append(diag_df)
464
465    # CASE_ATTRIBUTES: Static trace-level data
466    if feature_type in [config.PredictionFeature.CASE_ATTRIBUTES, config.PredictionFeature.HYBRID]:
467        # None => default to all available case attributes; [] => explicitly none.
468        cols = selected_case_attributes if selected_case_attributes is not None else list_case_attributes(log)
469        cols = [c for c in cols if c in log.columns and c != case_id_key]
470
471        if cols:
472            # Case attributes are static per case -> take the first observed value per case.
473            ca_df = log.groupby(case_id_key)[cols].first()
474            # Numeric attributes absent for some cases would feed NaN into StandardScaler;
475            # fill with 0. Categorical NaN is handled by get_dummies (becomes all-zero).
476            num_cols = ca_df.select_dtypes(include="number").columns
477            ca_df[num_cols] = ca_df[num_cols].fillna(0)
478            # Prefix mirrors cf_act_ to avoid naming collisions in the hybrid concat.
479            ca_df.columns = [f"ca_{c}" for c in ca_df.columns]
480        else:
481            # Fallback if the log has no (selected) case attributes.
482            ca_df = pd.DataFrame(index=log[case_id_key].unique())
483
484        feature_matrices.append(ca_df)
485
486    if not feature_matrices:
487        raise ValueError(f"Unsupported feature type: {feature_type}")
488        
489    # Concatenate side-by-side; all matrices share the case_id index, so pandas aligns them.
490    final_df = pd.concat(feature_matrices, axis=1)
491
492    return final_df

495def preprocess_for_ml(
496    X_raw: pd.DataFrame,
497    y_raw: pd.Series,
498    is_regression: bool = False,
499) -> Tuple[pd.DataFrame, np.ndarray, List[str]]:
500    """Encodes and scales the raw data for ML training.
501
502    Args:
503        X_raw (pd.DataFrame): Raw feature matrix.
504        y_raw (pd.Series): Raw label vector.
505        is_regression (bool, optional): When True the target is continuous (e.g. remaining 
506            time in seconds) — it is kept numeric and NOT label-encoded. Defaults to False.
507
508    Returns:
509        Tuple[pd.DataFrame, np.ndarray, List[str]]: A tuple containing:
510            - X: feature DataFrame (columns preserved for downstream alignment).
511            - y: encoded label vector (classification) or float vector (regression).
512            - class_names: ordered list of original class labels for classification, [] for regression.
513    """
514
515    # Feature Preprocessing (X)
516    # Converts categorical Case Attributes (strings) into numeric One-Hot encoded columns.
517    # Numerical features (Control Flow/Diagnostics/temporal) remain unchanged.
518    if X_raw.empty or len(X_raw.columns) == 0:
519        X = X_raw.copy()
520    else:
521        X = pd.get_dummies(X_raw, drop_first=True)
522
523    # Empty / missing labels: nothing to encode.
524    if y_raw is None or (hasattr(y_raw, "empty") and y_raw.empty):
525        return X, np.array([]), []
526
527    # Regression target (REMAINING_TIME): keep the continuous value, no encoding.
528    if is_regression:
529        y = np.asarray(y_raw, dtype=float)
530        return X, y, []

 49    log: pd.DataFrame,
 50    min_length: int = 1,
 51    max_length: Optional[int] = None,
 52    step: int = 1,
 53    gate_activities: Optional[List[str]] = None,
 54    case_id_key: str = "case:concept:name",
 55    timestamp_key: str = "time:timestamp",
 56    activity_key: str = "concept:name",
 57) -> Tuple[pd.DataFrame, pd.DataFrame]:
 58    """Expand each case into varying-length prefix pseudo-cases (LL-07.1).
 59
 60    For each case and each k in range(min_length, max_length+1, step) with k <= case length,
 61    the first k events (time-ordered) become a pseudo-case whose id is f"{orig}@@k{k}".
 62    This lets all existing feature/alignment/label code (which keys off case_id_key) run
 63    unchanged — each prefix simply looks like another case.
 64
 65    Leakage gating: if gate_activities is given (PROCESS_OUTCOME), only prefixes that end
 66    strictly before the first occurrence of any gate activity are emitted, so the outcome is
 67    never already inside the prefix.
 68
 69    Args:
 70        log (pd.DataFrame): The original event log.
 71        min_length (int, optional): Minimum prefix length. Defaults to 1.
 72        max_length (Optional[int], optional): Maximum prefix length. Defaults to None.
 73        step (int, optional): Step size between prefix lengths. Defaults to 1.
 74        gate_activities (Optional[List[str]], optional): Activities that gate prefix generation. Defaults to None.
 75        case_id_key (str, optional): Key for case ID. Defaults to "case:concept:name".
 76        timestamp_key (str, optional): Key for timestamp. Defaults to "time:timestamp".
 77        activity_key (str, optional): Key for activity name. Defaults to "concept:name".
 78
 79    Returns:
 80        Tuple[pd.DataFrame, pd.DataFrame]: A tuple containing:
 81            prefix_log: event-level DataFrame, case_id_key replaced by the unique prefix id,
 82                every other column (events, case attributes, timestamp) preserved.
 83            prefix_index: DataFrame [prefix_id, orig_case_id, prefix_length].
 84    """
 85    # Order events within each case; mergesort keeps original order for equal timestamps.
 86    g = log.sort_values([case_id_key, timestamp_key], kind="mergesort").copy()
 87    g["_ev"] = g.groupby(case_id_key).cumcount()          # 0-based event position
 88    case_len = g.groupby(case_id_key)["_ev"].max() + 1     # Series indexed by case id
 89
 90    # Per-case cap on prefix length. With gating, cap at the position of the first outcome
 91    # event: a length-k prefix covers events 0..k-1, so k <= first_pos excludes the outcome.
 92    if gate_activities:
 93        is_gate = g[activity_key].isin(set(gate_activities))
 94        gate_pos = g.loc[is_gate].groupby(case_id_key)["_ev"].min()
 95        cap = case_len.copy()
 96        cap.loc[gate_pos.index] = gate_pos
 97    else:
 98        cap = case_len
 99
100    # Build the set of valid (case, k) pairs.
101    specs = []
102    for case_id, c_cap in cap.items():
103        hi = int(c_cap) if max_length is None else min(int(max_length), int(c_cap))
104        for k in range(min_length, hi + 1, step):
105            specs.append((case_id, k))
106
107    if not specs:
108        return (
109            pd.DataFrame(columns=list(log.columns)),
110            pd.DataFrame(columns=["prefix_id", "orig_case_id", "prefix_length"]),
111        )
112
113    spec = pd.DataFrame(specs, columns=[case_id_key, "_k"])
114
115    # Expand: each event joins every valid k of its case, keep events strictly inside the prefix.
116    merged = g.merge(spec, on=case_id_key)
117    merged = merged[merged["_ev"] < merged["_k"]].copy()
118
119    merged["orig_case_id"] = merged[case_id_key]
120    merged[case_id_key] = merged[case_id_key].astype(str) + "@@k" + merged["_k"].astype(str)
121
122    prefix_index = (
123        merged[[case_id_key, "orig_case_id", "_k"]]
124        .drop_duplicates()
125        .rename(columns={case_id_key: "prefix_id", "_k": "prefix_length"})
126        .reset_index(drop=True)
127    )
128    # Keep prefix_length out of the event log so it can never leak into the feature matrix.
129    prefix_log = merged.drop(columns=["_ev", "_k", "orig_case_id"]).reset_index(drop=True)
130    return prefix_log, prefix_index

211def real_deviation_cost(alignment_moves: List[Tuple[Any, Any]]) -> int:
212    """Count the REAL conformance deviations in one alignment.
213
214    A pm4py alignment move is a (log_move, model_move) tuple:
215        (a, a)        synchronous move                                         -> 0 (conforming)
216        (a, '>>')     log move: event a not allowed by the model here         -> +1
217        ('>>', a)     model move on a VISIBLE transition a (expected, missing) -> +1
218        ('>>', None)  model move on an invisible/silent (tau) transition       -> 0 (routing)
219
220    Silent/tau moves are normal routing in (e.g.) Inductive-Miner models and carry zero
221    cost; counting them as deviations is exactly the bug this helper avoids. Mirrors the
222    move classification used for DIAGNOSTICS features in extract_features.
223
224    Args:
225        alignment_moves (List[Tuple[Any, Any]]): List of alignment moves.
226
227    Returns:
228        int: Total cost of real deviations.
229    """
230    cost = 0
231    for log_move, model_move in alignment_moves:
232        if log_move == model_move and log_move != '>>':
233            continue                                   # synchronous move
234        elif model_move == '>>' and log_move is not None:
235            cost += 1                                  # real log move (a, '>>')
236        elif log_move == '>>' and model_move is not None:
237            cost += 1                                  # real visible model move ('>>', a)
238        # else: invisible/tau model move ('>>', None) -> not a deviation
239    return cost

250

255

262

266

410        alignments (Optional[List[Dict[str, Any]]], optional): Alignment results. Defaults to None.
411        selected_case_attributes (Optional[List[str]], optional): Subset of case attributes. Defaults to None.
412
413    Raises:
414        ValueError: If required data for feature type is missing or if feature type is unsupported.
415
416    Returns:
417        pd.DataFrame: Feature matrix indexed by case ID.
418    """
419    case_id_key = 'case:concept:name'
420    activity_key = 'concept:name'
421    
422    feature_matrices = []
423    
424    # CONTROL_FLOW: Bag-of-Activities
425    if feature_type in [config.PredictionFeature.CONTROL_FLOW, config.PredictionFeature.HYBRID]:
426        # pd.crosstab -> frequency matrix (rows=cases, cols=activities)
427        cf_df = pd.crosstab(log[case_id_key], log[activity_key])
428        
429        # Prefix columns to avoid naming collisions with case attributes later
430        cf_df.columns = [f"cf_act_{col}" for col in cf_df.columns]
431        feature_matrices.append(cf_df)
432
433    # DIAGNOSTICS: Bag-of-[activity x move_type]
434    if feature_type in [config.PredictionFeature.DIAGNOSTICS, config.PredictionFeature.HYBRID]:
435        if alignments is None:
436            raise ValueError(f"{feature_type.name} requires alignment results.")
437            
438        case_ids = log[case_id_key].unique()
439        if len(case_ids) != len(alignments):
440            raise ValueError("Number of alignments does not match unique cases in log.")
441            
442        diag_records = []
443        for case_id, align_dict in zip(case_ids, alignments):
444            record = {'case_id': case_id}
445            alignment_moves = align_dict.get('alignment', [])
446            
447            for log_move, model_move in alignment_moves:
448                # Determine the specific move type and activity
449                if log_move == model_move and log_move != '>>':
450                    feat_name = f"sync_move_{log_move}"
451                elif model_move == '>>' and log_move != None:
452                    feat_name = f"log_move_{log_move}"
453                elif log_move == '>>' and model_move != None:
454                    feat_name = f"model_move_{model_move}"
455                else:
456                    # Invisible transitions
457                    continue 
458                
459                record[feat_name] = record.get(feat_name, 0) + 1
460                
461            diag_records.append(record)
462
463        diag_df = pd.DataFrame(diag_records).set_index('case_id').fillna(0).astype(int)
464        feature_matrices.append(diag_df)
465
466    # CASE_ATTRIBUTES: Static trace-level data
467    if feature_type in [config.PredictionFeature.CASE_ATTRIBUTES, config.PredictionFeature.HYBRID]:
468        # None => default to all available case attributes; [] => explicitly none.
469        cols = selected_case_attributes if selected_case_attributes is not None else list_case_attributes(log)
470        cols = [c for c in cols if c in log.columns and c != case_id_key]
471
472        if cols:
473            # Case attributes are static per case -> take the first observed value per case.
474            ca_df = log.groupby(case_id_key)[cols].first()
475            # Numeric attributes absent for some cases would feed NaN into StandardScaler;
476            # fill with 0. Categorical NaN is handled by get_dummies (becomes all-zero).
477            num_cols = ca_df.select_dtypes(include="number").columns
478            ca_df[num_cols] = ca_df[num_cols].fillna(0)
479            # Prefix mirrors cf_act_ to avoid naming collisions in the hybrid concat.
480            ca_df.columns = [f"ca_{c}" for c in ca_df.columns]
481        else:
482            # Fallback if the log has no (selected) case attributes.
483            ca_df = pd.DataFrame(index=log[case_id_key].unique())
484
485        feature_matrices.append(ca_df)
486
487    if not feature_matrices:
488        raise ValueError(f"Unsupported feature type: {feature_type}")
489        
490    # Concatenate side-by-side; all matrices share the case_id index, so pandas aligns them.
491    final_df = pd.concat(feature_matrices, axis=1)
492
493    return final_df

Extracts features based on the specified feature type.

  • CONTROL_FLOW: Extracts the trace prefix (Bag-of-activities).
  • DIAGNOSTICS: Extracts the alignment-based trace prefix diagnostics (Bag-of-move-types).
  • CASE_ATTRIBUTES: Extracts static attributes about the case.
  • HYBRID: Combines the active feature sets.
Arguments:
  • log (pd.DataFrame): Event log.
  • feature_type (PredictionFeature): Type of features to extract.
  • model (Optional[Tuple], optional): Process model. Defaults to None.
  • alignments (Optional[List[Dict[str, Any]]], optional): Alignment results. Defaults to None.
  • selected_case_attributes (Optional[List[str]], optional): Subset of case attributes. Defaults to None.
Raises:
  • ValueError: If required data for feature type is missing or if feature type is unsupported.
Returns:

pd.DataFrame: Feature matrix indexed by case ID.

516    # Feature Preprocessing (X)
517    # Converts categorical Case Attributes (strings) into numeric One-Hot encoded columns.
518    # Numerical features (Control Flow/Diagnostics/temporal) remain unchanged.
519    if X_raw.empty or len(X_raw.columns) == 0:
520        X = X_raw.copy()
521    else:
522        X = pd.get_dummies(X_raw, drop_first=True)

527
528    # Regression target (REMAINING_TIME): keep the continuous value, no encoding.

531        return X, y, []

533    # Classification target (PROCESS_OUTCOME / FUTURE_DEVIATIONS):
534    # map string/categorical labels to integers (0, 1, 2, ...).
535    le = LabelEncoder()
536    y = le.fit_transform(y_raw)
537    class_names = [str(c) for c in le.classes_]
